const CORS_HEADERS = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS',
  'Access-Control-Allow-Headers': 'Content-Type, X-Admin-Key, X-User-Id',
  'Content-Type': 'application/json'
};

const ADMIN_SECRET_KEY = "HabibulAdmin@2025";

// Cache TTLs (in seconds)
const CACHE_TTL = {
  FILES: 300,           // 5 minutes
  CATEGORIES: 600,      // 10 minutes
  NOTIFICATIONS: 300,   // 5 minutes
  QUICK_ACCESS: 600,    // 10 minutes
  SLIDER_CARDS: 600,    // 10 minutes
  MOCK_TESTS: 600,      // 10 minutes
  QUIZZES: 600          // 10 minutes
};

function generateId() {
  return Date.now().toString(36) + Math.random().toString(36).substring(2, 10);
}

function isAdminRequest(request) {
  const adminKey = request.headers.get('X-Admin-Key');
  return adminKey === ADMIN_SECRET_KEY;
}

// ============================================
// CACHE HELPER FUNCTIONS
// ============================================

async function getCachedResponse(cacheKey, ttlSeconds, fetchFunction) {
  const cache = caches.default;
  const cacheUrl = new URL(`https://cache.onlibry.com/${cacheKey}`);
  
  const cachedResponse = await cache.match(cacheUrl);
  
  if (cachedResponse) {
    console.log(`📦 Cache HIT: ${cacheKey}`);
    return cachedResponse;
  }
  
  console.log(`💾 Cache MISS: ${cacheKey}, fetching from DB...`);
  
  const data = await fetchFunction();
  
  const response = new Response(JSON.stringify(data), {
    headers: {
      ...CORS_HEADERS,
      'Cache-Control': `public, max-age=${ttlSeconds}`,
      'CF-Cache-Status': 'MISS'
    }
  });
  
  await cache.put(cacheUrl, response.clone());
  
  return response;
}

async function invalidateCache(cacheKey) {
  const cache = caches.default;
  const cacheUrl = new URL(`https://cache.onlibry.com/${cacheKey}`);
  await cache.delete(cacheUrl);
  console.log(`🗑️ Cache invalidated: ${cacheKey}`);
}

async function invalidateMultipleCaches(keys) {
  for (const key of keys) {
    await invalidateCache(key);
  }
}

    // PUT /api/admin/categories/:id - Invalidate cache
    if (method === 'PUT' && url.pathname.startsWith('/api/admin/categories/')) {
      const id = url.pathname.split('/api/admin/categories/')[1];
      const body = await request.json();
      const now = new Date().toISOString();
      await db.prepare(`UPDATE categories SET name = ?, slug = ?, color = ?, icon = ?, "order" = ?, updated_at = ? WHERE id = ?`).bind(body.name, body.slug, body.color, body.icon, body.order || 0, now, id).run();
      await invalidateCache('categories_all');
      return new Response(JSON.stringify({ success: true }), { headers: CORS_HEADERS });
    }
    
    // DELETE /api/admin/categories/:id - Invalidate cache
    if (method === 'DELETE' && url.pathname.startsWith('/api/admin/categories/')) {
      const id = url.pathname.split('/api/admin/categories/')[1];
      await db.prepare(`DELETE FROM categories WHERE id = ?`).bind(id).run();
      await invalidateCache('categories_all');
      return new Response(JSON.stringify({ success: true }), { headers: CORS_HEADERS });
    }
    
    // POST /api/admin/notifications - Invalidate cache
    if (method === 'POST' && url.pathname === '/api/admin/notifications') {
      const body = await request.json();
      const id = generateId();
      const now = new Date().toISOString();
      await db.prepare(`INSERT INTO notifications (id, title, content, category_id, link, image_url, is_pinned, status, published_at, created_at, updated_at) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`).bind(id, body.title, body.content, body.category_id, body.link, body.image_url, body.is_pinned || 0, body.status || 'published', now, now, now).run();
      await invalidateMultipleCaches(['notifications_latest_5', 'notifications_latest_10', 'notifications_latest_15']);
      return new Response(JSON.stringify({ success: true, id }), { headers: CORS_HEADERS });
    }
    
    // PUT /api/admin/notifications/:id - Invalidate cache
    if (method === 'PUT' && url.pathname.startsWith('/api/admin/notifications/')) {
      const id = url.pathname.split('/api/admin/notifications/')[1];
      const body = await request.json();
      const now = new Date().toISOString();
      await db.prepare(`UPDATE notifications SET title = ?, content = ?, category_id = ?, link = ?, image_url = ?, is_pinned = ?, status = ?, updated_at = ? WHERE id = ?`).bind(body.title, body.content, body.category_id, body.link, body.image_url, body.is_pinned || 0, body.status || 'published', now, id).run();
      await invalidateMultipleCaches(['notifications_latest_5', 'notifications_latest_10', 'notifications_latest_15']);
      return new Response(JSON.stringify({ success: true }), { headers: CORS_HEADERS });
    }
    
    // DELETE /api/admin/notifications/:id - Invalidate cache
    if (method === 'DELETE' && url.pathname.startsWith('/api/admin/notifications/')) {
      const id = url.pathname.split('/api/admin/notifications/')[1];
      await db.prepare(`DELETE FROM notifications WHERE id = ?`).bind(id).run();
      await invalidateMultipleCaches(['notifications_latest_5', 'notifications_latest_10', 'notifications_latest_15']);
      return new Response(JSON.stringify({ success: true }), { headers: CORS_HEADERS });
    }
    
    // POST /api/admin/slider-cards - Invalidate cache
    if (method === 'POST' && url.pathname === '/api/admin/slider-cards') {
      const body = await request.json();
      const id = generateId();
      const now = new Date().toISOString();
      await db.prepare(`INSERT INTO slider_cards (id, title, description, image_url, link, button_text, "order", is_active, created_at, updated_at) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`).bind(id, body.title, body.description, body.image_url, body.link, body.button_text || 'View More', body.order || 0, 1, now, now).run();
      await invalidateCache('slider_cards_all');
      return new Response(JSON.stringify({ success: true, id }), { headers: CORS_HEADERS });
    }
    
    // PUT /api/admin/slider-cards/:id - Invalidate cache
    if (method === 'PUT' && url.pathname.startsWith('/api/admin/slider-cards/')) {
      const id = url.pathname.split('/api/admin/slider-cards/')[1];
      const body = await request.json();
      const now = new Date().toISOString();
      await db.prepare(`UPDATE slider_cards SET title = ?, description = ?, image_url = ?, link = ?, button_text = ?, "order" = ?, updated_at = ? WHERE id = ?`).bind(body.title, body.description, body.image_url, body.link, body.button_text, body.order || 0, now, id).run();
      await invalidateCache('slider_cards_all');
      return new Response(JSON.stringify({ success: true }), { headers: CORS_HEADERS });
    }
    
    // DELETE /api/admin/slider-cards/:id - Invalidate cache
    if (method === 'DELETE' && url.pathname.startsWith('/api/admin/slider-cards/')) {
      const id = url.pathname.split('/api/admin/slider-cards/')[1];
      await db.prepare(`DELETE FROM slider_cards WHERE id = ?`).bind(id).run();
      await invalidateCache('slider_cards_all');
      return new Response(JSON.stringify({ success: true }), { headers: CORS_HEADERS });
    }
    
    // ============================================
    // 11. FILE SYNC (Admin) - Invalidate file cache
    // ============================================
    
    if (method === 'POST' && url.pathname === '/api/sync/files') {
      const body = await request.json();
      const files = body.files || [];
      let synced = 0;
      let updated = 0;
      const now = new Date().toISOString();
      
      for (const file of files) {
        if (!file.id) continue;
        
        const existing = await db.prepare(`SELECT id FROM files WHERE id = ?`).bind(file.id).first();
        
        if (existing) {
          await db.prepare(`UPDATE files SET name = ?, size = ?, mime_type = ?, price = ?, is_premium = ?, show_on_website = ?, tags = ?, cloudflare_key = ?, updated_at = ? WHERE id = ?`).bind(file.name || '', file.size || 0, file.mimeType || 'application/pdf', file.price || 29, file.isPremium ? 1 : 0, file.showOnWebsite ? 1 : 0, file.tags || '', file.cloudflareKey || file.id, now, file.id).run();
          updated++;
        } else {
          await db.prepare(`INSERT INTO files (id, name, size, mime_type, price, is_premium, show_on_website, tags, cloudflare_key, created_at, updated_at) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`).bind(file.id, file.name || '', file.size || 0, file.mimeType || 'application/pdf', file.price || 29, file.isPremium ? 1 : 0, file.showOnWebsite ? 1 : 0, file.tags || '', file.cloudflareKey || file.id, now, now).run();
          synced++;
        }
      }
      
      await invalidateCache('files_page1');
      
      return new Response(JSON.stringify({ success: true, synced: synced, updated: updated, total: files.length }), { headers: CORS_HEADERS });
    }
    
    // ============================================
    // 12. SYNC USERS
    // ============================================
    
    if (method === 'POST' && url.pathname === '/api/sync/users') {
      const body = await request.json();
      const users = body.users || [];
      let synced = 0;
      let updated = 0;
      const now = new Date().toISOString();
      
      for (const user of users) {
        if (!user.id) continue;
        
        const existing = await db.prepare(`SELECT id FROM users WHERE id = ?`).bind(user.id).first();
        
        if (existing) {
          await db.prepare(`UPDATE users SET email = ?, display_name = ?, photo_url = ?, is_admin = ?, is_banned = ?, subscription_type = ?, subscription_end = ?, updated_at = ? WHERE id = ?`).bind(user.email || '', user.display_name || '', user.photo_url || '', user.is_admin || 0, user.is_banned || 0, user.subscription_type || 'free', user.subscription_end || null, now, user.id).run();
          updated++;
        } else {
          await db.prepare(`INSERT INTO users (id, email, display_name, photo_url, is_admin, is_banned, subscription_type, subscription_end, created_at, updated_at) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`).bind(user.id, user.email || '', user.display_name || '', user.photo_url || '', user.is_admin || 0, user.is_banned || 0, user.subscription_type || 'free', user.subscription_end || null, user.created_at || now, now).run();
          synced++;
        }
      }
      
      return new Response(JSON.stringify({ success: true, synced: synced, updated: updated, total: users.length }), { headers: CORS_HEADERS });
    }
    
    if (method === 'GET' && url.pathname === '/api/admin/users') {
      const limit = parseInt(url.searchParams.get('limit') || '500');
      const offset = parseInt(url.searchParams.get('offset') || '0');
      const search = url.searchParams.get('search') || '';
      
      let query = `SELECT * FROM users`;
      let params = [];
      
      if (search) {
        query += ` WHERE email LIKE ? OR display_name LIKE ?`;
        params.push(`%${search}%`, `%${search}%`);
      }
      
      query += ` ORDER BY created_at DESC LIMIT ? OFFSET ?`;
      params.push(limit, offset);
      
      const result = await db.prepare(query).bind(...params).all();
      
      let countQuery = `SELECT COUNT(*) as total FROM users`;
      let countParams = [];
      if (search) {
        countQuery = `SELECT COUNT(*) as total FROM users WHERE email LIKE ? OR display_name LIKE ?`;
        countParams.push(`%${search}%`, `%${search}%`);
      }
      const totalResult = await db.prepare(countQuery).bind(...countParams).first();
      
      return new Response(JSON.stringify({ success: true, users: result.results, pagination: { total: totalResult?.total || 0, limit: limit, offset: offset } }), { headers: CORS_HEADERS });
    }
    
    if (method === 'GET' && url.pathname === '/api/admin/payment-logs') {
      const limit = parseInt(url.searchParams.get('limit') || '200');
      const result = await db.prepare(`SELECT * FROM payment_logs ORDER BY timestamp DESC LIMIT ?`).bind(limit).all();
      return new Response(JSON.stringify({ success: true, logs: result.results }), { headers: CORS_HEADERS });
    }
    
    // 404
    return new Response(JSON.stringify({ error: 'Not found' }), { status: 404, headers: CORS_HEADERS });
  }
