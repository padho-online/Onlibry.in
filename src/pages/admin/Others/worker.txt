// GET /api/cart/:userId - Get user's active cart items
if (method === 'GET' && url.pathname.startsWith('/api/cart/')) {
  const userId = url.pathname.split('/api/cart/')[1];
  if (!userId) {
    return new Response(JSON.stringify({ success: false, error: 'User ID required' }), { status: 400, headers: CORS_HEADERS });
  }
  
  try {
    const result = await db.prepare(`
      SELECT file_id, file_name, price, item_type, action, created_at 
      FROM cart_logs 
      WHERE user_id = ? AND status = 'active'
      ORDER BY created_at DESC
    `).bind(userId).all();
    
    const cartMap = new Map();
    for (const log of result.results) {
      if (!cartMap.has(log.file_id)) {
        cartMap.set(log.file_id, {
          id: log.file_id,
          name: log.file_name,
          price: log.price,
          type: log.item_type || 'file',
          action: log.action,
          addedAt: log.created_at
        });
      }
    }
    
    const cart = Array.from(cartMap.values());
    console.log(`📦 Cart loaded for ${userId}: ${cart.length} items`);
    
    return new Response(JSON.stringify({ success: true, cart: cart }), { headers: CORS_HEADERS });
  } catch (error) {
    return new Response(JSON.stringify({ success: false, error: error.message }), { status: 500, headers: CORS_HEADERS });
  }
}

          INSERT INTO users (id, email, display_name, photo_url, is_admin, is_banned, subscription_type, subscription_end, created_at, updated_at)
          VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        `).bind(
          user.id,
          user.email || '',
          user.display_name || '',
          user.photo_url || '',
          user.is_admin ? 1 : 0,
          user.is_banned ? 1 : 0,
          user.subscription_type || '',
          user.subscription_end || '',
          user.created_at || now,
          now
        ).run();
        synced++;
      }
    }
    
    return new Response(JSON.stringify({ 
      success: true, 
      synced: synced, 
      updated: updated,
      total: users.length 
    }), { headers: CORS_HEADERS });
    
  } catch (error) {
    console.error('❌ Users sync error:', error);
    return new Response(JSON.stringify({ success: false, error: error.message }), { status: 500, headers: CORS_HEADERS });
  }
}

// GET /api/admin/users - Get all users from D1 (for admin panel)
if (method === 'GET' && url.pathname === '/api/admin/users') {
  if (!isAdminRequest(request)) {
    return new Response(JSON.stringify({ success: false, error: 'Unauthorized' }), { status: 401, headers: CORS_HEADERS });
  }
  
  try {
    const limit = parseInt(url.searchParams.get('limit') || '100');
    const offset = parseInt(url.searchParams.get('offset') || '0');
    const search = url.searchParams.get('search') || '';
    
    let query = `SELECT * FROM users`;
    let params = [];
    
    if (search) {
      query += ` WHERE email LIKE ? OR display_name LIKE ?`;
      params.push(`%${search}%`, `%${search}%`);
    }
    
    query += ` ORDER BY created_at DESC LIMIT ? OFFSET ?`;
    params.push(limit, offset);
    
    const result = await db.prepare(query).bind(...params).all();
    
    // Get total count
    let countQuery = `SELECT COUNT(*) as total FROM users`;
    let countParams = [];
    if (search) {
      countQuery = `SELECT COUNT(*) as total FROM users WHERE email LIKE ? OR display_name LIKE ?`;
      countParams.push(`%${search}%`, `%${search}%`);
    }
    const totalResult = await db.prepare(countQuery).bind(...countParams).first();
    
    return new Response(JSON.stringify({ 
      success: true, 
      users: result.results,
      pagination: {
        total: totalResult?.total || 0,
        limit: limit,
        offset: offset
      }
    }), { headers: CORS_HEADERS });
  } catch (error) {
    return new Response(JSON.stringify({ success: false, error: error.message }), { status: 500, headers: CORS_HEADERS });
  }
}
    
    // 404
    return new Response(JSON.stringify({ error: 'Not found' }), { status: 404, headers: CORS_HEADERS });
  }
};
