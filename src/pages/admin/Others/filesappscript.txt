// ============================================
// HELPER: Create JSON response
// ============================================
function createResponse(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

// ============================================
// DO GET - Handle ALL GET requests
// ============================================
function doGet(e) {
  const action = e?.parameter?.action;
  
  if (action === "getEditorData") return getEditorData();
  if (action === "getFolders") return getFolders();
  if (action === "add") return addFileViaGet(e.parameter);
  if (action === "update") return updateFileViaGet(e.parameter);
  if (action === "delete") return deleteFileViaGet(e.parameter);
  if (action === "clearEditor") return clearEditor();
  if (action === "bulkUpdate") return bulkUpdateFromEditor({});
  
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(MAIN_SHEET_NAME);
  if (!sheet) return createResponse({ success: false, error: "Sheet not found" });
  
  const data = getSheetDataOptimized(sheet);
  const rows = data.slice(1);
  const isAdminRequest = e?.parameter?.admin === "true";
  const files = [];
  
  for (let i = 0; i < rows.length; i++) {
    const row = rows[i];
    const status = row[12];
    if (status === "deleted") continue;
    
    if (!isAdminRequest) {
      const showOnWebsite = row[6] === true || row[6] === "TRUE";
      if (!showOnWebsite) continue;
    }
    
    let tagsObject = {};
    let tagsString = "";
    
    if (row[7]) {
      tagsString = row[7].toString();
      const tagPairs = tagsString.split(',');
      tagPairs.forEach(pair => {
        const trimmedPair = pair.trim();
        if (trimmedPair.includes(':')) {
          const colonIndex = trimmedPair.indexOf(':');
          const key = trimmedPair.substring(0, colonIndex).trim();
          const value = trimmedPair.substring(colonIndex + 1).trim();
          if (key && value) {
            if (!tagsObject[key]) tagsObject[key] = [];
            tagsObject[key].push(value);
          }
        }
      });
    }
    
    files.push({
      id: row[0],
      name: row[1],
      size: row[2],
      mimeType: row[3],
      price: row[4],
      isPremium: row[5] === true || row[5] === "TRUE",
      showOnWebsite: row[6] === true || row[6] === "TRUE",
      tags: tagsObject,
      tagsString: tagsString,
      createdAt: row[8],
      cloudflareKey: row[9],
      actualFileLink: row[10],
      websiteViewLink: row[11]
    });
  }
  
  return createResponse({ success: true, files: files });
}

// ============================================
// OPTIMIZED: Get sheet data (faster)
// ============================================
function getSheetDataOptimized(sheet) {
  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();
  if (lastRow <= 1) return [];
  return sheet.getRange(1, 1, lastRow, lastCol).getValues();
}

// ============================================
// DO POST - Handle POST requests
// ============================================
function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const action = data.action;
    
    switch(action) {
      case "add": return addFile(data);
      case "update": return updateFile(data);
      case "delete": return deleteFile(data);
      case "log": return addLog(data);
      case "bulkUpdate": return bulkUpdateFromEditor(data);
      case "getEditorData": return getEditorData();
      case "clearEditor": return clearEditor();
      case "incrementView": return incrementViewCount(data);
      case "incrementDownload": return incrementDownloadCount(data);
      default: return createResponse({ success: false, message: "Unknown action: " + action });
    }
  } catch (error) {
    return createResponse({ success: false, error: error.toString() });
  }
}

// GET EDITOR DATA
// ============================================
function getEditorData() {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(EDITOR_SHEET_NAME);
    
    if (!sheet) {
      const newSheet = SpreadsheetApp.getActiveSpreadsheet().insertSheet(EDITOR_SHEET_NAME);

      return createResponse({ success: true, files: [] });
    }
    
    const data = getSheetDataOptimized(sheet);
    
    if (!data || data.length <= 1) {
      return createResponse({ success: true, files: [] });
    }
    
    const rows = data.slice(1);
    const validRows = [];
    
    for (let i = 0; i < rows.length; i++) {
      const row = rows[i];
      const fileId = row[0];
      
      if (fileId && fileId.toString().trim() !== "" && fileId !== "fileId") {
        validRows.push({
          fileId: row[0] ? row[0].toString() : "",
          fileName: row[1] ? row[1].toString() : "",
          fileSize: row[2] ? row[2].toString() : "",
          mimeType: row[3] ? row[3].toString() : "",
          price: row[4] !== undefined && row[4] !== null && row[4].toString().trim() !== "" ? row[4].toString() : "",
          isPremium: row[5] !== undefined && row[5] !== null && row[5].toString().trim() !== "" ? row[5].toString() : "",
          showOnWebsite: row[6] !== undefined && row[6] !== null && row[6].toString().trim() !== "" ? row[6].toString() : "",
          tags: row[7] !== undefined && row[7] !== null && row[7].toString().trim() !== "" ? row[7].toString() : "",
          status: row[8] !== undefined && row[8] !== null && row[8].toString().trim() !== "" ? row[8].toString().toLowerCase() : ""
        });
      }
    }
    
    return createResponse({ success: true, files: validRows });
    
  } catch (error) {
    Logger.log(`❌ Error in getEditorData: ${error.toString()}`);
    return createResponse({ success: false, error: error.toString() });
  }
}

// UPDATE SINGLE FILE (NO AUTO SYNC)
// ============================================
function updateFile(data) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(MAIN_SHEET_NAME);
  const range = sheet.getDataRange();
  const values = range.getValues();
  
  let changes = [];
  let found = false;
  let rowIndex = -1;
  
  for (let i = 1; i < values.length; i++) {
    if (values[i][0] === data.fileId) {
      rowIndex = i;
      found = true;
      break;
    }
  }
  
  if (!found) {
    return createResponse({ success: false, error: "File not found: " + data.fileId });
  }
  
  if (data.hasOwnProperty('fileName') && data.fileName !== undefined && data.fileName !== null && data.fileName !== '') {
    const oldValue = values[rowIndex][1];
    if (data.fileName !== oldValue) {
      sheet.getRange(rowIndex + 1, 2).setValue(data.fileName);
      changes.push(`name: "${oldValue}" → "${data.fileName}"`);
    }
  }
  
  if (data.hasOwnProperty('price') && data.price !== undefined && !isNaN(data.price)) {
    const oldValue = values[rowIndex][4];
    if (data.price !== oldValue) {
      sheet.getRange(rowIndex + 1, 5).setValue(data.price);
      changes.push(`price: ₹${oldValue} → ₹${data.price}`);
    }
  }
  
  if (data.hasOwnProperty('isPremium') && data.isPremium !== undefined && typeof data.isPremium === 'boolean') {
    const oldValue = values[rowIndex][5];
    if (data.isPremium !== oldValue) {
      sheet.getRange(rowIndex + 1, 6).setValue(data.isPremium);
      changes.push(`premium: ${oldValue ? "Yes" : "No"} → ${data.isPremium ? "Yes" : "No"}`);
    }
  }
  
  if (data.hasOwnProperty('showOnWebsite') && data.showOnWebsite !== undefined && typeof data.showOnWebsite === 'boolean') {
    const oldValue = values[rowIndex][6];
    if (data.showOnWebsite !== oldValue) {
      sheet.getRange(rowIndex + 1, 7).setValue(data.showOnWebsite);
      changes.push(`visibility: ${oldValue ? "Visible" : "Hidden"} → ${data.showOnWebsite ? "Visible" : "Hidden"}`);
    }
  }
  
  if (data.hasOwnProperty('tags') && data.tags !== undefined && data.tags !== null && data.tags !== '') {
    const oldValue = values[rowIndex][7];
    if (data.tags !== oldValue) {
      sheet.getRange(rowIndex + 1, 8).setValue(data.tags);
      changes.push(`tags: "${oldValue}" → "${data.tags}"`);
    }
  }
  
  if (changes.length > 0) {
    addLog({
      fileId: data.fileId,
      fileName: data.fileName || values[rowIndex][1],
      action: "edit",
      performedBy: data.updatedBy,
      performedByName: data.updatedByName,
      changes: changes.join(" | "),
      ipAddress: data.userIp,
      userAgent: data.userAgent
    });
  }
  
  return createResponse({ success: true, message: "File updated", changes: changes });
}

// BULK UPDATE FROM EDITOR SHEET
// ============================================
function bulkUpdateFromEditor(data) {
  try {
    const editorSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(EDITOR_SHEET_NAME);
    
    if (!editorSheet) {
      return createResponse({ success: false, error: "Editor sheet not found" });
    }
    
    const editorData = getSheetDataOptimized(editorSheet);
    
    if (editorData.length <= 1) {
      return createResponse({ success: false, error: "No data in Editor sheet" });
    }
    
    const rows = editorData.slice(1);
    const mainSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(MAIN_SHEET_NAME);
    const mainValues = getSheetDataOptimized(mainSheet);
    
    let updatedCount = 0;
    let errors = [];
    let allChanges = [];
    
    function hasValue(cell) {
      if (cell === undefined || cell === null) return false;
      if (typeof cell === 'boolean') return true;
      const str = cell.toString().trim();
      return str !== "";
    }
    
    for (let i = 0; i < rows.length; i++) {
      const editorRow = rows[i];
      const fileId = editorRow[0] ? editorRow[0].toString().trim() : "";
      
      if (!fileId) {
        errors.push(`Row ${i + 2}: No fileId provided`);
        continue;
      }
      
      let mainRowIndex = -1;
      for (let j = 1; j < mainValues.length; j++) {
        if (mainValues[j][0] && mainValues[j][0].toString() === fileId) {
          mainRowIndex = j;
          break;
        }
      }
      
      if (mainRowIndex === -1) {
        errors.push(`Row ${i + 2}: File ID "${fileId}" not found`);
        continue;
      }
      
      let changes = [];
      
      if (hasValue(editorRow[1])) {
        const newValue = editorRow[1].toString().trim();
        const oldValue = mainValues[mainRowIndex][1];
        if (newValue !== oldValue) {
          mainSheet.getRange(mainRowIndex + 1, 2).setValue(newValue);
          changes.push(`name: "${oldValue}" → "${newValue}"`);
        }
      }
      
      if (hasValue(editorRow[2])) {
        const newValue = editorRow[2].toString().trim();
        const oldValue = mainValues[mainRowIndex][2];
        if (newValue !== oldValue) {
          mainSheet.getRange(mainRowIndex + 1, 3).setValue(newValue);
          changes.push(`size: "${oldValue}" → "${newValue}"`);
        }
      }
      
      if (hasValue(editorRow[3])) {
        const newValue = editorRow[3].toString().trim();
        const oldValue = mainValues[mainRowIndex][3];
        if (newValue !== oldValue) {
          mainSheet.getRange(mainRowIndex + 1, 4).setValue(newValue);
          changes.push(`mimeType: "${oldValue}" → "${newValue}"`);
        }
      }
      
      if (hasValue(editorRow[4])) {
        const newValue = Number(editorRow[4]);
        const oldValue = mainValues[mainRowIndex][4];
        if (!isNaN(newValue) && newValue !== oldValue) {
          mainSheet.getRange(mainRowIndex + 1, 5).setValue(newValue);
          changes.push(`price: ₹${oldValue} → ₹${newValue}`);
        }
      }
      
      if (hasValue(editorRow[5])) {
        const val = editorRow[5].toString().toLowerCase().trim();
        const newValue = (val === 'true' || val === 'yes' || val === '1');
        const oldValue = mainValues[mainRowIndex][5];
        if (newValue !== oldValue) {
          mainSheet.getRange(mainRowIndex + 1, 6).setValue(newValue);
          changes.push(`premium: ${oldValue ? "Yes" : "No"} → ${newValue ? "Yes" : "No"}`);
        }
      }
      
      if (hasValue(editorRow[6])) {
        const val = editorRow[6].toString().toLowerCase().trim();
        let newValue = (val === 'true' || val === 'yes' || val === '1');
        const oldValue = mainValues[mainRowIndex][6];
        if (newValue !== oldValue) {
          mainSheet.getRange(mainRowIndex + 1, 7).setValue(newValue);
          changes.push(`visibility: ${oldValue ? "Visible" : "Hidden"} → ${newValue ? "Visible" : "Hidden"}`);
        }
      }
      
      if (hasValue(editorRow[7])) {
        const newValue = editorRow[7].toString().trim();
        const oldValue = mainValues[mainRowIndex][7];
        if (newValue !== oldValue) {
          mainSheet.getRange(mainRowIndex + 1, 8).setValue(newValue);
          changes.push(`tags: "${oldValue}" → "${newValue}"`);
        }
      }
      
      if (hasValue(editorRow[8])) {
        const newValue = editorRow[8].toString().toLowerCase().trim();
        const oldValue = mainValues[mainRowIndex][12];
        if (newValue !== oldValue) {
          mainSheet.getRange(mainRowIndex + 1, 13).setValue(newValue);
          changes.push(`status: "${oldValue}" → "${newValue}"`);
        }
      }
      
      if (changes.length > 0) {
        updatedCount++;
        allChanges.push({ fileId: fileId, changes: changes });
        addLog({
          fileId: fileId,
          fileName: hasValue(editorRow[1]) ? editorRow[1] : mainValues[mainRowIndex][1],
          action: "bulk_edit",
          performedBy: data?.updatedBy || "system",
          performedByName: data?.updatedByName || "Bulk Update",
          changes: changes.join(" | "),
          ipAddress: data?.userIp || "",
          userAgent: data?.userAgent || ""
        });
      }
    }
    
    const lastRow = editorSheet.getLastRow();
    if (lastRow > 1) {
      editorSheet.deleteRows(2, lastRow - 1);
    }
    
    return createResponse({ 
      success: true, 
      message: `Updated ${updatedCount} files`, 
      updatedCount: updatedCount,
      changes: allChanges,
      errors: errors.length > 0 ? errors : undefined
    });
    
  } catch (error) {
    Logger.log(`❌ Error in bulkUpdateFromEditor: ${error.toString()}`);
    return createResponse({ success: false, error: error.toString() });
  }
}

// DELETE FILE (Soft delete) - NO AUTO SYNC
// ============================================
function deleteFile(data) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(MAIN_SHEET_NAME);
  const range = sheet.getDataRange();
  const values = range.getValues();
  
  let found = false;
  
  for (let i = 1; i < values.length; i++) {
    if (values[i][0] === data.fileId) {
      sheet.getRange(i+1, 13).setValue("deleted");
      found = true;
      break;
    }
  }
  
  if (!found) {
    return createResponse({ success: false, error: "File not found: " + data.fileId });
  }
  
  addLog({
    fileId: data.fileId,
    fileName: data.fileName,
    action: "delete",
    performedBy: data.deletedBy,
    performedByName: data.deletedByName,
    changes: `File deleted. Reason: ${data.reason || "Manual delete"}`,
    ipAddress: data.userIp,
    userAgent: data.userAgent
  });
  
  return createResponse({ success: true, message: "File deleted" });
}

// CLEAR EDITOR SHEET
// ============================================
function clearEditor() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(EDITOR_SHEET_NAME);
  if (!sheet) {
    return createResponse({ success: false, error: "Editor sheet not found" });
  }
  
  const lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    sheet.deleteRows(2, lastRow - 1);
  }
  
  return createResponse({ success: true, message: "Editor sheet cleared" });
}

// INCREMENT VIEW COUNT
// ============================================
function incrementViewCount(data) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(STATS_SHEET_NAME);
  if (!sheet) return;
  
  const values = getSheetDataOptimized(sheet);
  
  for (let i = 1; i < values.length; i++) {
    if (values[i][0] === data.fileId) {
      const currentCount = Number(values[i][2]) || 0;
      sheet.getRange(i+1, 3).setValue(currentCount + 1);
      sheet.getRange(i+1, 6).setValue(new Date().toISOString());
      break;
    }
  }
}

      sheet.getRange(i+1, 4).setValue(currentCount + 1);
      sheet.getRange(i+1, 7).setValue(new Date().toISOString());
      break;
    }
  }
}

function syncChangedFilesOnly() {
  const startTime = new Date();
  let syncType = "INCREMENTAL";
  let synced = 0, updated = 0, deleted = 0;
  let status = "SUCCESS";
  let errorMsg = "";
  
  try {
    console.log('🔄 Starting INCREMENTAL sync (only changed files)...');
    
    const lastSyncTime = getLastSyncTime();
    
    if (!lastSyncTime) {
      console.log('⚠️ First sync - running FULL sync instead');
      const result = fullSyncToD1();
      addSyncStatsEntry("FULL", result.synced, result.updated, result.deleted, result.success ? "SUCCESS" : "FAILED", 
        (new Date() - startTime) / 1000, result.error || "");
      return result;
    }
    
    console.log(`📅 Last sync: ${lastSyncTime}`);
    
    const ss = SpreadsheetApp.openById(SHEET_ID);
    const sheet = ss.getSheetByName(D1_SHEET_NAME);
    if (!sheet) throw new Error(`Sheet '${D1_SHEET_NAME}' not found`);
    
    const data = getSheetDataOptimized(sheet);
    const headers = data[0];
    const rows = data.slice(1);
    
    const findCol = (possibleNames) => {
      for (const name of possibleNames) {
        const index = headers.findIndex(h => String(h || '').toLowerCase().trim() === name.toLowerCase().trim());
        if (index !== -1) return index;
      }
      return -1;
    };
    
    const idCol = findCol(['fileId', 'id', 'file_id']);
    const nameCol = findCol(['fileName', 'name', 'file_name']);
    const sizeCol = findCol(['fileSize', 'size', 'file_size']);
    const mimeCol = findCol(['mimeType', 'mime_type']);
    const priceCol = findCol(['price']);
    const premiumCol = findCol(['isPremium', 'is_premium', 'premium']);
    const showCol = findCol(['showOnWebsite', 'show_on_website', 'visible']);
    const tagsCol = findCol(['tags', 'tag']);
    const cloudflareCol = findCol(['cloudflareKey', 'cloudflare_key']);
    const createdAtCol = findCol(['createdAt', 'created_at', 'uploadedAt']);
    const statusCol = findCol(['status']);
    const updatedAtCol = findCol(['updatedAt', 'updated_at', 'lastModified']);
    
    const changedFiles = [];
    const deletedFileIds = [];
    
    for (let i = 0; i < rows.length; i++) {
      const row = rows[i];
      if (!row || row.length === 0) continue;
      
      const fileStatus = statusCol !== -1 ? String(row[statusCol] || '').toLowerCase() : '';
      
      let fileId = '';
      if (idCol !== -1 && row[idCol]) fileId = String(row[idCol]).trim();
      else if (cloudflareCol !== -1 && row[cloudflareCol]) fileId = String(row[cloudflareCol]).trim();
      else if (row[0]) fileId = String(row[0]).trim();
      
      if (!fileId) continue;
      
      let needsSync = false;
      
      if (fileStatus === 'deleted') {
        needsSync = true;
        deletedFileIds.push(fileId);
      } else {
        if (updatedAtCol !== -1 && row[updatedAtCol]) {
          try {
            const fileModifiedTime = new Date(row[updatedAtCol]);
            if (fileModifiedTime > lastSyncTime) {
              needsSync = true;
            }
          } catch(e) {}
        }
        
        if (!needsSync && createdAtCol !== -1 && row[createdAtCol]) {
          try {
            const fileCreatedTime = new Date(row[createdAtCol]);
            if (fileCreatedTime > lastSyncTime) {
              needsSync = true;
            }
          } catch(e) {}
        }
      }
      
      if (needsSync && fileStatus !== 'deleted') {
        let showOnWebsite = true;
        if (showCol !== -1) {
          const val = row[showCol];
          showOnWebsite = (val === true || val === 'TRUE' || val === 'true' || val === '1');
        }
        
        let isPremium = false;
        if (premiumCol !== -1) {
          const val = row[premiumCol];
          isPremium = (val === true || val === 'TRUE' || val === 'true' || val === '1');
        }
        
        let price = 29;
        if (priceCol !== -1 && row[priceCol]) {
          const p = Number(row[priceCol]);
          if (!isNaN(p)) price = p;
        }
        
        let size = 0;
        if (sizeCol !== -1 && row[sizeCol]) {
          const s = Number(row[sizeCol]);
          if (!isNaN(s)) size = s;
        }
        
        changedFiles.push({
          id: fileId,
          name: nameCol !== -1 ? String(row[nameCol] || '').trim() : '',
          size: size,
          mimeType: mimeCol !== -1 ? String(row[mimeCol] || 'application/pdf') : 'application/pdf',
          price: price,
          isPremium: isPremium,
          showOnWebsite: showOnWebsite,
          tags: tagsCol !== -1 ? String(row[tagsCol] || '') : '',
          cloudflareKey: cloudflareCol !== -1 ? String(row[cloudflareCol] || '') : fileId,
          createdAt: createdAtCol !== -1 && row[createdAtCol] ? String(row[createdAtCol]) : new Date().toISOString()
        });
      }
    }
    
    console.log(`📊 Changed files: ${changedFiles.length}, Deleted: ${deletedFileIds.length}`);
    
    if (changedFiles.length === 0 && deletedFileIds.length === 0) {
      console.log('✅ No changes detected since last sync');
      updateLastSyncTime();
      addSyncStatsEntry(syncType, 0, 0, 0, "NO_CHANGES", (new Date() - startTime) / 1000, "");
      return { success: true, message: 'No changes', synced: 0, updated: 0, deleted: 0 };
    }
    
    const BATCH_SIZE = 100;
    for (let i = 0; i < changedFiles.length; i += BATCH_SIZE) {
      const batch = changedFiles.slice(i, i + BATCH_SIZE);
      const batchNum = Math.floor(i / BATCH_SIZE) + 1;
      const totalBatches = Math.ceil(changedFiles.length / BATCH_SIZE);
      
      console.log(`📤 Sending incremental batch ${batchNum}/${totalBatches}: ${batch.length} files`);
      
      try {
        const options = {
          method: 'POST',
          headers: { 'Content-Type': 'application/json', 'X-Admin-Key': ADMIN_KEY },
          muteHttpExceptions: true,
          payload: JSON.stringify({ files: batch })
        };
        
        const response = UrlFetchApp.fetch(`${D1_API_URL}/api/sync/files`, options);
        const result = JSON.parse(response.getContentText());
        
        synced += result.synced || 0;
        updated += result.updated || 0;
        
        console.log(`✅ Incremental batch ${batchNum} complete: +${result.synced} new, ~${result.updated} updated`);
        
        Utilities.sleep(200);
      } catch (err) {
        console.error(`❌ Incremental batch ${batchNum} failed: ${err.toString()}`);
        errorMsg = err.toString();
      }
    }
    
    for (const fileId of deletedFileIds) {
      try {
        const options = {
          method: 'DELETE',
          headers: { 'X-Admin-Key': ADMIN_KEY },
          muteHttpExceptions: true
        };
        const response = UrlFetchApp.fetch(`${D1_API_URL}/api/admin/files/${encodeURIComponent(fileId)}`, options);
        const result = JSON.parse(response.getContentText());
        if (result.success) deleted++;
      } catch (err) {
        console.error(`Failed to delete ${fileId}: ${err.toString()}`);
      }
    }
    
    updateLastSyncTime();
    
    const duration = (new Date() - startTime) / 1000;
    console.log(`🎉 INCREMENTAL SYNC COMPLETE: +${synced} new, ~${updated} updated, -${deleted} deleted in ${duration}s`);
    
    addSyncStatsEntry(syncType, synced, updated, deleted, status, duration, errorMsg || "");
    
    return { success: true, synced: synced, updated: updated, deleted: deleted };
    
  } catch (error) {
    status = "FAILED";
    errorMsg = error.toString();
    console.error(`❌ Incremental sync error: ${errorMsg}`);
    addSyncStatsEntry(syncType, synced, updated, deleted, status, (new Date() - startTime) / 1000, errorMsg);
    return { success: false, error: errorMsg };
  }
}

function continueSync() {
  const progress = getSyncProgress();
  if (progress.resumeIndex > 0 && progress.resumeIndex < progress.totalFiles) {
    console.log(`🔄 Resuming sync from ${progress.resumeIndex}/${progress.totalFiles}`);
    return syncFilesToD1();
  }
  return { success: false, message: 'No sync in progress' };
}
